● Create one service.yaml file with type nodeport & select frontend-webapp pod with port
80 & targetPort 80 with any nodePort between range 30000-32768.

# service.yaml
apiVersion: v1
kind: Service
metadata:
  name: flask-webapp-service
spec:
  selector:
    app: frontend-webapp
  ports:
    - protocol: TCP
      port: 80
      targetPort: 80
      nodePort: 30008
  type: NodePort

alam@DESKTOP-VK19M09:kube$ kubectl apply -f service.yaml
